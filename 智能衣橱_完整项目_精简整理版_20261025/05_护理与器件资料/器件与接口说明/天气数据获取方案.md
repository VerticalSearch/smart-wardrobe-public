# 单片机获取天气数据方案

## 一、方案概述

### 1.1 实现方式

**硬件**：ESP8266 WiFi模块 + STM32F407VGT6
**通信**：UART串口通信（AT指令）
**数据源**：天气API（如心知天气、和风天气、OpenWeatherMap等）

### 1.2 工作流程

```
STM32 → ESP8266 → WiFi → 天气API服务器 → JSON数据 → ESP8266 → STM32
```

## 二、硬件连接

### 2.1 ESP8266连接

**ESP8266模块（ESP-01S）引脚**：

| 引脚 | 功能 | 连接STM32 |
|------|------|----------|
| VCC | 3.3V电源 | 3.3V（注意：ESP8266需要3.3V，不能接5V） |
| GND | 地 | GND |
| TX | 串口发送 | STM32的RX（PA10） |
| RX | 串口接收 | STM32的TX（PA9） |
| CH_PD | 使能（高电平有效） | 3.3V或GPIO控制 |
| RST | 复位（低电平复位） | GPIO控制（可选） |

**STM32F407VGT6连接（UART1）**：
```
ESP8266          STM32F407VGT6
VCC       →      3.3V
GND       →      GND
TX        →      PA10 (UART1_RX)
RX        →      PA9  (UART1_TX)
CH_PD     →      3.3V（或PB0 GPIO控制）
RST       →      PB1（可选，GPIO控制）
```

### 2.2 电源注意事项

**重要**：ESP8266必须使用3.3V供电，不能接5V，否则会烧毁！

**供电方案**：
- 方案1：STM32板上的3.3V输出（如果电流足够）
- 方案2：独立的3.3V LDO模块（推荐，更稳定）
- 方案3：USB转串口模块的3.3V输出（调试时）

## 三、通信协议

### 3.1 AT指令

ESP8266通过AT指令控制，STM32发送AT指令，ESP8266返回响应。

**基本AT指令**：
```
AT              // 测试通信
AT+RST          // 复位模块
AT+GMR          // 查看版本信息
AT+CWMODE=1     // 设置为Station模式（连接WiFi）
AT+CWJAP="SSID","密码"  // 连接WiFi
AT+CIPSTART="TCP","api.seniverse.com",80  // 连接服务器
AT+CIPSEND=长度  // 发送数据
```

### 3.2 HTTP请求

**GET请求格式**：
```
GET /v3/weather/now.json?key=你的密钥&location=城市&language=zh-Hans HTTP/1.1
Host: api.seniverse.com
Connection: close

```

## 五、STM32代码实现

### 5.1 UART配置

```c
// UART1配置（ESP8266，标准库）
void UART1_Init(void) {
    GPIO_InitTypeDef GPIO_InitStructure;
    USART_InitTypeDef USART_InitStructure;
    NVIC_InitTypeDef NVIC_InitStructure;
    
    // 使能时钟
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_USART1 | RCC_APB2Periph_GPIOA | RCC_APB2Periph_AFIO, ENABLE);
    
    // 配置TX引脚（PA9）
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_9;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AF_PP;  // 复用推挽输出
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(GPIOA, &GPIO_InitStructure);
    
    // 配置RX引脚（PA10）
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_10;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_IN_FLOATING;  // 浮空输入
    GPIO_Init(GPIOA, &GPIO_InitStructure);
    
    // 配置UART
    USART_InitStructure.USART_BaudRate = 115200;  // ESP8266默认115200
    USART_InitStructure.USART_WordLength = USART_WordLength_8b;
    USART_InitStructure.USART_StopBits = USART_StopBits_1;
    USART_InitStructure.USART_Parity = USART_Parity_No;
    USART_InitStructure.USART_HardwareFlowControl = USART_HardwareFlowControl_None;
    USART_InitStructure.USART_Mode = USART_Mode_Rx | USART_Mode_Tx;
    USART_Init(USART1, &USART_InitStructure);
    
    // 使能UART
    USART_Cmd(USART1, ENABLE);
    
    // 使能接收中断
    USART_ITConfig(USART1, USART_IT_RXNE, ENABLE);
    
    // 配置NVIC
    NVIC_InitStructure.NVIC_IRQChannel = USART1_IRQn;
    NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 1;
    NVIC_InitStructure.NVIC_IRQChannelSubPriority = 1;
    NVIC_InitStructure.NVIC_IRQChannelCmd = ENABLE;
    NVIC_Init(&NVIC_InitStructure);
}
```

### 5.2 AT指令发送函数

```c
// 发送一个字节（标准库）
void UART1_SendByte(uint8_t byte) {
    while (USART_GetFlagStatus(USART1, USART_FLAG_TXE) == RESET);  // 等待发送缓冲区空
    USART_SendData(USART1, byte);
    while (USART_GetFlagStatus(USART1, USART_FLAG_TC) == RESET);  // 等待发送完成
}

// 发送字符串（标准库）
void UART1_SendString(char *str) {
    while (*str) {
        UART1_SendByte(*str++);
    }
}

// 发送AT指令（标准库）
void ESP8266_SendCmd(char *cmd) {
    UART1_SendString(cmd);
    UART1_SendString("\r\n");  // 发送回车换行
    Delay_ms(100);  // 等待响应
}

// 发送AT指令并等待响应（标准库）
uint8_t ESP8266_SendCmd_Wait(char *cmd, char *response, uint16_t timeout) {
    ESP8266_SendCmd(cmd);
    
    uint32_t start_time = GetTickCount();  // 标准库获取系统时间
    while (GetTickCount() - start_time < timeout) {
        if (strstr((char *)rx_buffer, response) != NULL) {
            return 1;  // 找到响应
        }
        Delay_ms(10);
    }
    return 0;  // 超时
}
```

### 5.3 WiFi连接

```c
// 连接WiFi
uint8_t ESP8266_ConnectWiFi(char *ssid, char *password) {
    // 1. 测试AT指令
    if (!ESP8266_SendCmd_Wait("AT", "OK", 1000)) {
        return 0;  // 通信失败
    }
    
    // 2. 设置为Station模式
    if (!ESP8266_SendCmd_Wait("AT+CWMODE=1", "OK", 1000)) {
        return 0;
    }
    
    // 3. 连接WiFi
    char cmd[128];
    sprintf(cmd, "AT+CWJAP=\"%s\",\"%s\"", ssid, password);
    if (!ESP8266_SendCmd_Wait(cmd, "OK", 10000)) {  // WiFi连接可能需要较长时间
        return 0;
    }
    
    // 4. 查询IP地址（可选）
    ESP8266_SendCmd("AT+CIFSR");
    
    return 1;  // 连接成功
}
```

### 5.4 HTTP请求天气数据

```c
// 获取天气数据
uint8_t ESP8266_GetWeather(char *api_key, char *city, WeatherData_t *weather) {
    // 1. 建立TCP连接
    char cmd[128];
    sprintf(cmd, "AT+CIPSTART=\"TCP\",\"api.seniverse.com\",80");
    if (!ESP8266_SendCmd_Wait(cmd, "OK", 5000)) {
        return 0;  // 连接失败
    }
    
    Delay_ms(500);
    
    // 2. 构造HTTP请求
    char http_request[256];
    sprintf(http_request, 
        "GET /v3/weather/now.json?key=%s&location=%s&language=zh-Hans HTTP/1.1\r\n"
        "Host: api.seniverse.com\r\n"
        "Connection: close\r\n"
        "\r\n",
        api_key, city);
    
    // 3. 发送数据长度
    sprintf(cmd, "AT+CIPSEND=%d", strlen(http_request));
    ESP8266_SendCmd(cmd);
    Delay_ms(100);
    
    // 4. 发送HTTP请求（标准库）
    UART1_SendString(http_request);
    Delay_ms(2000);  // 等待服务器响应
    
    // 5. 接收响应数据
    // 数据在rx_buffer中，需要解析JSON
    
    // 6. 关闭连接
    ESP8266_SendCmd("AT+CIPCLOSE");
    
    return 1;
}
```

### 5.5 JSON数据解析

```c
// 简单的JSON解析（提取温度）
float Parse_Temperature(char *json_data) {
    // 查找 "temperature":"25"
    char *temp_str = strstr(json_data, "\"temperature\":\"");
    if (temp_str == NULL) {
        return -999;  // 未找到
    }
    
    temp_str += 16;  // 跳过 "temperature":"
    float temp = atof(temp_str);  // 转换为浮点数
    
    return temp;
}

// 解析湿度
float Parse_Humidity(char *json_data) {
    char *hum_str = strstr(json_data, "\"humidity\":\"");
    if (hum_str == NULL) {
        return -999;
    }
    
    hum_str += 13;  // 跳过 "humidity":"
    float humidity = atof(hum_str);
    
    return humidity;
}

// 解析天气状况
void Parse_WeatherText(char *json_data, char *weather_text) {
    char *text_str = strstr(json_data, "\"text\":\"");
    if (text_str == NULL) {
        strcpy(weather_text, "未知");
        return;
    }
    
    text_str += 8;  // 跳过 "text":"
    int i = 0;
    while (text_str[i] != '"' && i < 15) {
        weather_text[i] = text_str[i];
        i++;
    }
    weather_text[i] = '\0';
}
```

### 5.6 完整流程

```c
// 天气数据结构
typedef struct {
    float temperature;      // 温度
    float humidity;         // 湿度
    char weather_text[16];  // 天气状况（晴、雨等）
    float wind_speed;      // 风速
    uint8_t valid;         // 数据有效标志
} WeatherData_t;

WeatherData_t weather_data;

// 获取天气数据（主函数，标准库）
void Get_WeatherData(void) {
    // 1. 初始化ESP8266
    UART1_Init();
    Delay_ms(1000);
    
    // 2. 连接WiFi
    if (!ESP8266_ConnectWiFi("你的WiFi名称", "WiFi密码")) {
        printf("WiFi连接失败\n");
        return;
    }
    printf("WiFi连接成功\n");
    
    // 3. 获取天气数据
    if (ESP8266_GetWeather("你的API密钥", "beijing", &weather_data)) {
        // 4. 解析JSON数据
        weather_data.temperature = Parse_Temperature((char *)rx_buffer);
        weather_data.humidity = Parse_Humidity((char *)rx_buffer);
        Parse_WeatherText((char *)rx_buffer, weather_data.weather_text);
        weather_data.valid = 1;
        
        printf("温度: %.1f°C\n", weather_data.temperature);
        printf("湿度: %.1f%%\n", weather_data.humidity);
        printf("天气: %s\n", weather_data.weather_text);
    } else {
        printf("获取天气数据失败\n");
        weather_data.valid = 0;
    }
}

## 八、常见问题

### 8.1 ESP8266无响应

**问题**：发送AT指令无响应
- 检查电源是否为3.3V（不能接5V）
- 检查波特率是否为115200
- 检查TX/RX是否接反
- 检查CH_PD是否接高电平

### 8.2 WiFi连接失败

**问题**：无法连接WiFi
- 检查WiFi名称和密码是否正确
- 检查WiFi信号强度
- 增加等待时间（WiFi连接可能需要10-15秒）
- 检查ESP8266固件版本

### 8.3 HTTP请求失败

**问题**：无法获取天气数据
- 检查API密钥是否正确
- 检查城市名称是否正确
- 检查网络连接是否正常
- 检查HTTP请求格式是否正确

### 8.4 JSON解析失败

**问题**：无法解析天气数据
- 检查返回的JSON数据格式
- 使用串口打印查看原始数据
- 检查JSON解析函数是否正确

        if (ESP8266_GetWeather(api_key, city, weather)) {

## 十一、总结

通过ESP8266获取天气数据的步骤：
1. **硬件连接**：ESP8266通过UART连接到STM32
2. **WiFi连接**：使用AT指令连接WiFi
3. **HTTP请求**：发送HTTP GET请求到天气API
4. **数据解析**：解析返回的JSON数据
5. **数据使用**：用于智能通风、穿衣推荐等功能

**关键点**：
- ESP8266必须使用3.3V供电
- AT指令需要以"\r\n"结尾
- HTTP请求格式要正确
- JSON解析需要根据实际数据格式调整
- 注意API调用频率限制
